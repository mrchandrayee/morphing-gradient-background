CSS
.morphing-bg {
  width: 750px;
  height: 300px;
  background: linear-gradient(45deg, #ee7752, #e73c7e, #23a6d5, #23d5ab);
  background-size: 400% 400%;
  animation: gradient 8s ease infinite, morph 5s ease infinite;
  border-radius: 20px;
  transform-style: preserve-3d;
  box-shadow: 5px 15px 20px 0 rgba(0, 0, 0, 0.4);
  border: 2.5px solid #eee;
}

/* Animation keyframes added to make the displayed CSS runnable */
@keyframes gradient {
  0%   { background-position: 0% 50%; }
  50%  { background-position: 100% 50%; }
  100% { background-position: 0% 50%; }
}

@keyframes morph {
  0%, 100% { border-radius: 20px; transform: perspective(900px) rotateX(0deg) rotateY(0deg); }
  25%      { border-radius: 35% 20% 30% 18%; transform: perspective(900px) rotateX(2deg) rotateY(-3deg); }
  50%      { border-radius: 20px 38% 18% 32%; transform: perspective(900px) rotateX(-2deg) rotateY(3deg); }
  75%      { border-radius: 28% 18% 36% 20%; transform: perspective(900px) rotateX(1deg) rotateY(-2deg); }
}
